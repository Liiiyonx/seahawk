async def list_by_task(self, task_id, *, page=1, page_size=20) -> tuple[list[TaskAck], int]:
    count_stmt = select(func.count()).select_from(TaskAck).where(TaskAck.task_id == task_id)
    total = int((await self.session.execute(count_stmt)).scalar_one())
    stmt = (
        select(TaskAck)
        .where(TaskAck.task_id == task_id)
        .order_by(TaskAck.received_wall_at.desc())
        .limit(page_size)
        .offset((page - 1) * page_size)
    )
    rows = await self.session.execute(stmt)
    return list(rows.scalars().all()), total

class RacingSession:
    """模拟并发竞争窗口：首次 SELECT 未命中（对方未提交），flush 撞唯一键。"""
    def __init__(self, real, *, fail_flush: bool = True):
        self._real = real
        self._selects = 0
        self._fail_flush = fail_flush
    def __getattr__(self, name):
        return getattr(self._real, name)
    async def execute(self, stmt, *a, **kw):
        self._selects += 1
        if self._selects == 1 and self._fail_flush:
            # 竞争窗口：假装 SELECT 未命中
            return _EmptyResult()
        return await self._real.execute(stmt, *a, **kw)
    async def flush(self):
        if self._fail_flush:
            raise IntegrityError("INSERT", {}, Exception("unique_violation"))
        return await self._real.flush()

_RacingSession:
```python
class _RacingSession:
    """模拟并发首次到达竞争窗口：
    - 第 1 次 execute（SELECT 查重）假装未命中（对方事务尚未提交）；
    - 之后的 flush 走真实会话 → 真实撞唯一键（IntegrityError）；
    - rollback 后的重查命中对方已提交的规范行 → 转 duplicate。
    """
    def __init__(self, real):
        self._real = real
        self._selects = 0
    def __getattr__(self, name):
        return getattr(self._real, name)
    async def execute(self, stmt, *args, **kwargs):
        self._selects += 1
        if self._selects == 1:
            return _EmptyResult()
        return await self._real.execute(stmt, *args, **kwargs)

class _EmptyResult:
    def scalar_one_or_none(self):
        return None
```

Then `await self.session.flush()` → __getattr__ → real.flush → REAL IntegrityError (row exists, committed). → caught → `await self.session.rollback()` → real.rollback. → `get_by_command_id` → execute #2 → real query → finds committed row → _bump_duplicate → returns DUPLICATE. ✓✓.
