  def save_seq_atomic(path, seq) -> bool:
      """原子写盘：先写同目录临时文件再 os.replace 替换；失败返回 False 不抛异常。"""

              f.flush()
              try:

              except OSError:
                  pass
          os.replace(tmp, path)
          return True
      except OSError:

          except OSError:
              pass
          return False
