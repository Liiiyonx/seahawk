```python
"""平台侧 ACK 接收：冻结回执信封解析、身份校验与幂等判定（WP-14C）。"""

from __future__ import annotations

import threading
from dataclasses import dataclass, field
from datetime import datetime, timezone
from typing import Any, Callable

from loguru import logger


class AckResult:
    NEW = "new"
    DUPLICATE = "duplicate"
    OUT_OF_ORDER = "out_of_order"
    LATE = "late"
    ALL = (NEW, DUPLICATE, OUT_OF_ORDER, LATE)


@dataclass(frozen=True)
class AckRecord:
    """一条已登记（首次规范）回执。"""
    command_id: str
    task_id: str
    device_id: str
    seq: int
    received_at: float
    accepted: bool
    reason: str
    mode: str
    outcome: str
    ack_id: str = ""


def parse_received_at(value: Any) -> float | None:
    """解析 received_at（或旧字段 ts）为 epoch 秒；解析失败返回 None。

    兼容：float/int（epoch 秒）、ISO-8601 字符串（含 Z 后缀）、datetime。
    """
    if isinstance(value, bool):
        return None
    if isinstance(value, (int, float)):
        return float(value)
    if isinstance(value, datetime):
        return value.timestamp() if value.tzinfo else value.replace(tzinfo=timezone.utc).timestamp()
    if isinstance(value, str):
        text = value.strip()
        if text == "":
            return None
        try:
            dt = datetime.fromisoformat(text.replace("Z", "+00:00"))
            return dt.timestamp() if dt.tzinfo else dt.replace(tzinfo=timezone.utc).timestamp()
        except ValueError:
            # epoch 秒字符串也接受（孪生时间模型为 float）
            try:
                return float(text)
            except ValueError:
                return None
    return None
```

```python
class AckTracker:
    """平台侧 ACK 幂等跟踪（内存实现，进程内有效）。

    判定优先级：duplicate > late > out_of_order > new。

        self._deadlines: dict[str, float] = {}
        self._max_acked_seq: dict[str, int] = {}
        self._lock = threading.Lock()
        self.outcomes: list[str] = []

    def register_command(self, command_id, device_id, seq, expires_at) -> None:
        """发布成功后登记命令（供 late 判定）。仅进程内有效。"""
        self._deadlines[command_id] = float(expires_at)

    def forget_command(self, command_id) -> None:  # maybe for cleanup? Not required.
    
    def record(self, ack: AckEnvelope) -> str:
        with self._lock:
            existing = self._acks.get(ack.command_id)
            if existing is not None:
                result = AckResult.DUPLICATE
            else:
```

```python
@dataclass(frozen=True)
class AckRecord:
    """一条回执的首次规范记录（duplicate 时原样返回）。"""
    command_id: str
    task_id: str
    device_id: str
    seq: int
    received_at: float
    accepted: bool
    reason: str
    mode: str
    outcome: str
    ack_id: str = ""


class AckTracker:
    """平台侧 ACK 幂等跟踪（进程内，内存实现）。

    与 edge/device_sim 的 AckTracker 语义一致（冻结契约），但**不 import**
    edge.device_sim —— 平台侧逻辑禁止反向依赖孪生包。

    登记与判定：
    - `register_command(...)`：**仅在 MQTT 发布成功后**由 client.publish_task 调用，
      登记 command_id/device_id/seq/expires_at（供 late 判定）。
      进程重启后登记表丢失 —— 未登记命令**仍可正常接收**（不伪判 late），
      这是有意的回落策略，不是持久化。
    - `record(ack)`：判定优先级 duplicate > late > out_of_order > new。
    """

    def __init__(self) -> None:
        self._acks: dict[str, AckRecord] = {}
        self._deadlines: dict[str, float] = {}
        self._max_acked_seq: dict[str, int] = {}
        self._lock = threading.Lock()
        self.outcomes: list[str] = []
        self.counts: dict[str, int] = {r: 0 for r in AckResult.ALL}

    def register_command(
        self,
        command_id: str,
        device_id: str,
        seq: int,
        expires_at: float,
    ) -> None:
        """登记已发布命令的过期时间（仅发布成功后调用）。"""
        with self._lock:
            self._deadlines[command_id] = float(expires_at)

    def has_deadline(self, command_id: str) -> bool:
        return command_id in self._deadlines

    def record(self, ack: AckEnvelope) -> str:
        """登记一条回执并返回判定结果；同 command_id 幂等。"""
        with self._lock:
            existing = self._acks.get(ack.command_id)
            if existing is not None:
                result = AckResult.DUPLICATE
            else:
                deadline = self._deadlines.get(ack.command_id)
                if deadline is not None and ack.received_at > deadline:
                    result = AckResult.LATE
                elif ack.seq < self._max_acked_seq.get(ack.device_id, ack.seq):
                    result = AckResult.OUT_OF_ORDER
                else:
                    self._max_acked_seq[ack.device_id] = max(
                        self._max_acked_seq.get(ack.device_id, -1), ack.seq
                    )
                    result = AckResult.NEW
                self._acks[ack.command_id] = AckRecord(
                    command_id=ack.command_id,
                    task_id=ack.task_id,
                    device_id=ack.device_id,
                    seq=ack.seq,
                    received_at=ack.received_at,
                    accepted=ack.accepted,
                    reason=ack.reason,
                    mode=ack.mode,
                    outcome=result,
                    ack_id=ack.ack_id,
                )
            self.outcomes.append(result)
            self.counts[result] += 1
            return result

    def is_acked(self, command_id: str) -> bool:
        return command_id in self._acks

    def ack_for(self, command_id: str) -> AckRecord | None:
        return self._acks.get(command_id)
```

For the NEXT new ack with seq=2: 2 < 1? No → NEW. Good. But there's a bug scenario: what if the first ack for a device is late (seq=5), watermark set to 5.     字段缺失或类型错误只记 warning 并丢弃，绝不让 MQTT 主循环抛异常。

```python
        envelope = parse_ack_envelope(payload)
        if envelope is None:

            first = tracker.ack_for(envelope.command_id)

                f"[MQTT-ACK] 重复回执（{envelope.command_id}），返回首次记录，"
                f"不重复推进（首次判定={first.outcome if first else '?'}）"
            )

            session_factory = get_session_factory

        async with session_factory()() as session:
            from app.repositories import TaskRepository

            task = await TaskRepository(session).get_by_task_id(envelope.task_id)
            if task is None:

                    f"[MQTT-ACK] 任务不存在，回执已记录但无法推进："
                    f"task={envelope.task_id} command={envelope.command_id} "
                    f"outcome={outcome}"

                    f"[MQTT-ACK] 设备身份不一致，整条拒绝（不回退不推进）："
                    f"payload.device_id={envelope.device_id} topic={topic_robot} "
                    f"task.robot_id={task.robot_id} task={task.task_id}"
                )
                return outcome

            engine = DispatchEngine(session)
            from app.models.task import TaskStatus

            if envelope.accepted:

                if task.status == TaskStatus.ASSIGNED:
                    await engine.transition(task, TaskStatus.NAVIGATING)
```

## 四、关键设计

- **重复先于身份校验短路**：重复 ACK 直接返回首次规范回执，不再碰 DB——否则 accepted=false 首次回退清空 `robot_id` 后，重复拒绝回执会因身份失配永远到不了 duplicate 分支。
- **被拒回执不登记**：未知任务/身份不一致的回执不写入 AckTracker，避免污染去重表导致后续合法回执被误判 duplicate（有专门测试锁定）。
- **落库成功后登记**：DB 状态机是权威，内存跟踪表是去重/审计层；进程重启丢表后，DB 的 assigned 守卫保证不重复推进。
- **不伪判 late**：late 判定仅在 `register_command` 已登记 deadline 时生效；未登记命令按 new/out_of_order 正常接收推进。
- **时钟可注入**：received_at 为报文携带（epoch/ISO 均可注入），`build_dispatch_payload(now=...)` 已有注入点；测试全用假时钟值，无 datetime.now 断言。
- 旧格式 `seq` 缺失兜底 0、`command_id` 由 `task_id` 派生——既满足"旧 ACK 字段仍兼容"矩阵项，又保证去重键统一。
